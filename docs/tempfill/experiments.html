tempfill.experiments

End-to-end experiments: audit, tuning, validation, performance, reconstruction.

Data protocol for the primary sensor (0869, chronological)::

|----------- development (60%) -----------|------ test (40%) ------|
 harmonic tuning, method comparison,         used once, after every
 hybrid switch and adoption decisions        decision is frozen

The secondary sensor (0326: different room, different year) is never used for any decision. It is a second independent test, and the case study for long gaps (up to 3 days) and its real multi-day outages.

Two artificial-gap designs (validation.sample_gaps):

  • fully observed - every slot in the gap holds a real reading; used for 10 min to 2 h, where such windows are common;
  • partially observed - at least 80% of slots hold real readings and only those are scored; used for 1 h and longer, because fully observed windows longer than a few hours are rare in this data.

Both designs are run at 1 h and 2 h so their agreement can be checked. Headline tables use the fully observed results up to 2 h and the partially observed results above.

Every stage writes CSV tables to results/tables and figures to results/figures; write_summary collects the key numbers in results/RESULTS.md.

120def method_set(hk: dict, switch_len: int | None = None) -> dict:
121    """All compared methods; harmonic uses the tuned settings ``hk``."""
122    methods = {
123        "ffill": M.fill_ffill, "linear": M.fill_linear, "yesterday": M.fill_yesterday,
124        "cubic_spline": M.fill_cubic_spline, "pchip": M.fill_pchip,

128        methods["hybrid"] = lambda f: M.fill_hybrid(f, switch_len=switch_len, **hk)

132def evaluate(frame, methods, region, cfg: Config, seed: int, long_lengths=None,
133             repeats_long=None) -> dict:
134    """Run both gap designs and build the headline (combined) summary.
135
136    Returns:
137        Dict with ``full``, ``partial`` and ``combined`` summaries (each with
138        a ``design`` column) and the two per-gap tables.
139    """
140    long_lengths = long_lengths or cfg.long_lengths
141    pg_f, pts_f = run_gap_experiment(frame, methods, cfg.short_lengths, cfg.n_gaps,
142                                     cfg.repeats_short, region, seed, min_observed=1.0,
143                                     keep_points=True)
144    pg_p, pts_p = run_gap_experiment(frame, methods, long_lengths, cfg.n_gaps,
145                                     repeats_long or cfg.repeats_long, region, seed + 1,
146                                     min_observed=0.8, keep_points=True)
147    full = summarise(pg_f, pts_f).assign(design="fully observed")
148    partial = summarise(pg_p, pts_p).assign(design="partially observed")
149    short_max = max(cfg.short_lengths)
150    combined = pd.concat([full, partial[partial["gap_len"] > short_max]], ignore_index=True)
151    return {"full": full, "partial": partial, "combined": combined,
152            "per_gap_full": pg_f, "per_gap_partial": pg_p}

158def audit(cfg: Config) -> Context:
159    """Load the data, audit every sensor and regularise the two usable ones."""
160    readings, raw_counts, load_stats = load_temperatures(cfg.root / cfg.data_zip)

165
166    ctx = Context(readings=readings, locations=locations, stats={"load": load_stats})
167    for sid in (cfg.primary, cfg.secondary):
168        frame, st = prepare_sensor(readings, sid)
169        ctx.frames[sid] = frame
170        ctx.stats[sid] = st
171    _save_table(pd.DataFrame([{"sensor_id": k, **v} for k, v in ctx.stats.items() if k != "load"]),
172                cfg, "prepared_series")
173
174    named = {f"Sensor {sid} ({locations.get(sid, 'location unknown')})": f
175             for sid, f in ctx.frames.items()}
176    plots.sensor_overview(named, cfg.figures / "01_sensor_overview.png")
177    plots.gap_length_histogram(named, cfg.figures / "02_gap_lengths.png")
178    return ctx

181# ----------------------------------------------------------------------------
182# Stage 2: tuning and method selection (development period only)
183# ----------------------------------------------------------------------------
184def tune(cfg: Config, ctx: Context) -> dict:
185    """Tune the harmonic model and apply the selection rules on development data.
186
187    Sweeps use the partially observed design at 3, 6 and 12 h, where the
188    harmonic settings matter, and ``min_len=6`` so real single dropouts are
189    not refitted each time (this does not affect the scored gaps).
190    """
191    frame = ctx.frames[cfg.primary]
192    dev, _ = _regions(len(frame), cfg)
193    sweep_lengths = (18, 36, 72)
194
195    def sweep(param, values, base):
196        tabs = []
197        for v in values:
198            kw = {**base, param: v}
199            pg = run_gap_experiment(frame, {v: (lambda f, kw=kw: M.fill_harmonic(f, min_len=6, **kw))},
200                                    sweep_lengths, cfg.n_gaps, cfg.tune_repeats, dev, cfg.seed)
201            tabs.append(summarise(pg).assign(**{param: v}))
202        tab = pd.concat(tabs, ignore_index=True)
203        best = tab.groupby(param)["MAE"].mean().idxmin()
204        return tab, best
205
206    k_tab, k = sweep("k_daily", (0, 1, 2, 3, 4, 6), {"k_weekly": 2, "window_days": 7.0})
207    j_tab, j = sweep("k_weekly", (0, 1, 2, 3), {"k_daily": int(k), "window_days": 7.0})
208    w_tab, w = sweep("window_days", (2.0, 3.0, 5.0, 7.0, 10.0, 14.0, 21.0),
209                     {"k_daily": int(k), "k_weekly": int(j)})
210    hk = {"k_daily": int(k), "k_weekly": int(j), "window_days": float(w)}
211    for name, tab in [("tune_k_daily", k_tab), ("tune_k_weekly", j_tab), ("tune_window", w_tab)]:
212        _save_table(tab.drop(columns="method"), cfg, name)
213    plots.sensitivity(k_tab, "k_daily", cfg.figures / "05_sens_k_daily.png",
214                      "Daily harmonics (development period)", "Number of daily harmonics K")
215    plots.sensitivity(w_tab, "window_days", cfg.figures / "06_sens_window.png",
216                      "Fitting window (development period)", "Window half-width (days)")
217
218    # Method comparison on development data, then the switch rule.
219    singles = evaluate(frame, method_set(hk), dev, cfg, cfg.seed + 10)
220    switch = choose_switch(singles["combined"], "linear", "harmonic", cfg.switch_tolerance)
221    # Same seeds -> same gap placements, so the hybrid is a paired comparison.
222    hyb = evaluate(frame, {"hybrid": lambda f: M.fill_hybrid(f, switch_len=switch, **hk)},
223                   dev, cfg, cfg.seed + 10)
224    dev_comb = pd.concat([singles["combined"], hyb["combined"]], ignore_index=True)
225    _save_table(dev_comb, cfg, "dev_method_comparison")
226
227    adopted, best_single = adopt_hybrid(dev_comb, "hybrid", cfg.resolution)
228    if adopted:
229        final = "hybrid"
230    else:
231        final = simplest_within(singles["combined"], M.COMPLEXITY_ORDER, cfg.switch_tolerance)
232
233    # Optimisation rule (stage 7): if the final method is the harmonic model,
234    # deploy it with short-gap skipping when that keeps accuracy within the
235    # switch tolerance at every length.
236    piv = dev_comb.pivot_table(index="gap_len", columns="method", values="MAE")
237    hybrid_ok = bool((piv["hybrid"] <= (1 + cfg.switch_tolerance) * piv["harmonic"]).all())
238    deployed = "hybrid" if final in ("harmonic", "hybrid") and hybrid_ok else final
239
240    score = dev_comb.groupby("method")["MAE"].mean().rename("mean_dev_MAE").reset_index()
241    params = {**hk, "switch_len": switch, "best_single": best_single,
242              "hybrid_adopted_on_accuracy": adopted, "final_method": final,
243              "hybrid_accuracy_matches_harmonic": hybrid_ok, "deployed": deployed,
244              "dev_scores": score.round(4).to_dict("records")}
245    _params_path(cfg).write_text(json.dumps(params, indent=2))
246    return params

250# Stage 3: held-out test and independent sensor
251# ----------------------------------------------------------------------------
252def validate(cfg: Config, ctx: Context, params: dict) -> dict:
253    """Final comparison on the held-out test period and on the second sensor."""
254    frame = ctx.frames[cfg.primary]
255    _, test = _regions(len(frame), cfg)
256    hk = _harmonic_kw(params)
257    methods = method_set(hk, params["switch_len"])
258
259    res = evaluate(frame, methods, test, cfg, cfg.seed + 20)
260    comb = _save_table(res["combined"], cfg, "test_results_primary")
261    _save_table(res["full"], cfg, "test_full_observed_primary")
262    _save_table(res["partial"], cfg, "test_partial_observed_primary")
263    _save_table(res["per_gap_partial"], cfg, "test_per_gap_partial_primary")
264
265    # Agreement of the two designs where both were run.
266    both = sorted(set(cfg.short_lengths) & set(cfg.long_lengths))
267    agree = (res["full"][res["full"]["gap_len"].isin(both)]
268             .merge(res["partial"][res["partial"]["gap_len"].isin(both)],
269                    on=["method", "gap_len", "gap_hours"], suffixes=("_full", "_partial")))
270    agree["CIs_overlap"] = ((agree["MAE_lo_full"] <= agree["MAE_hi_partial"])
271                            & (agree["MAE_lo_partial"] <= agree["MAE_hi_full"]))
272    agree = _save_table(agree[["method", "gap_hours", "MAE_full", "MAE_lo_full", "MAE_hi_full",
273                               "MAE_partial", "MAE_lo_partial", "MAE_hi_partial", "CIs_overlap"]],
274                        cfg, "design_agreement_primary")
275
276    horizon = usable_horizon(comb, cfg.acceptance_mae)
277    p95 = comb.pivot_table(index="gap_hours", columns="method", values="P95AE")
278    _save_table(horizon.rename_axis("method").reset_index(), cfg, "usable_horizon_primary")
279
280    plots.error_vs_gap(comb, cfg.figures / "03_error_vs_gap_test.png",
281                       f"Held-out test period, sensor {cfg.primary}", cfg.acceptance_mae,
282                       boundary=max(cfg.short_lengths) / 6)
283    plots.error_vs_gap(comb, cfg.figures / "03b_p95_vs_gap_test.png",
284                       f"95th-percentile error, sensor {cfg.primary}", cfg.p95_limit,
285                       metric="P95AE", boundary=max(cfg.short_lengths) / 6)
286    plots.design_agreement(agree, cfg.figures / "03c_design_agreement.png")
287    plots.error_by_hour(res["per_gap_partial"], frame, 36, ["linear", "pchip", "harmonic"],
288                        cfg.figures / "07_error_by_start_hour.png")
289
290    # Example 12-hour gap from the test period.
291    rng = np.random.default_rng(cfg.seed)
292    obs = frame["temperature"].notna().to_numpy()
293    start = sample_gaps(obs, 72, 1, rng, test)[0]
294    masked, _ = apply_gaps(frame, [start], 72)
295    fills = {m: methods[m](masked) for m in ("linear", "cubic_spline", "yesterday", "harmonic")}
296    plots.example_gap(frame, start, 72, fills, cfg.figures / "04_example_gap.png", context=216)
297
298    # Secondary sensor: never used for decisions; also the long-gap case study.
299    f2 = ctx.frames[cfg.secondary]
300    res2 = evaluate(f2, methods, None, cfg, cfg.seed + 30, long_lengths=cfg.case_lengths,
301                    repeats_long=max(2, cfg.repeats_long // 2))
302    comb2 = _save_table(res2["combined"], cfg, "test_results_secondary")
303    horizon2 = usable_horizon(comb2, cfg.acceptance_mae)
304    _save_table(horizon2.rename_axis("method").reset_index(), cfg, "usable_horizon_secondary")
305    plots.error_vs_gap(comb2, cfg.figures / "08_error_vs_gap_secondary.png",
306                       f"Independent sensor {cfg.secondary} (unseen room and year)",
307                       cfg.acceptance_mae, boundary=max(cfg.short_lengths) / 6)
308    return {"primary": comb, "secondary": comb2, "agreement": agree,
309            "horizon": horizon, "horizon2": horizon2, "p95": p95}

315def performance(cfg: Config, ctx: Context, params: dict) -> dict:
316    """Runtime, memory, optimisation steps, scaling, solver stability, profiling."""
317    frame = ctx.frames[cfg.primary]
318    hk = _harmonic_kw(params)
319    methods = method_set(hk, params["switch_len"])
320    reps = 1 if cfg.quick else 3
321
322    costs = P.method_costs(frame, methods, repeats=reps)
323    flops = P.flop_estimates(frame, switch_len=params["switch_len"], **hk)
324    costs = costs.merge(flops, on="method", how="left")
325    costs["MFLOP_per_s"] = (costs["flops"] / costs["seconds"] / 1e6).where(costs["flops"] > 0)
326    _save_table(costs, cfg, "perf_method_costs")
327
328    # Optimisation of the harmonic method, measured step by step. The
329    # secondary sensor is used as the workload too: it has many more and
330    # longer real gaps, so the differences are larger.
331    steps = {
332        "1 naive (basis rebuilt per gap)": lambda f: M.fill_harmonic_naive(f, **hk),
333        "2 basis built once": lambda f: M.fill_harmonic(f, **hk),
334        "3 + skip short gaps (hybrid)": lambda f: M.fill_hybrid(f, switch_len=params["switch_len"], **hk),
335    }
336    opt = pd.concat([P.optimisation_steps(ctx.frames[sid], steps, "1 naive (basis rebuilt per gap)",
337                                          repeats=reps).assign(sensor=sid)
338                     for sid in (cfg.primary, cfg.secondary)], ignore_index=True)
339    _save_table(opt, cfg, "perf_optimisation_steps")
340
341    big = P.synthetic_frame(100_000 if cfg.quick else 1_000_000)
342    vec = P.vectorisation_speedup(big, M.fill_linear_loop, M.fill_linear, repeats=reps)
343    solver = _save_table(P.solver_comparison(frame, **hk), cfg, "perf_solver_comparison")
344
345    sizes = (4_000, 16_000, 64_000) if cfg.quick else (4_000, 16_000, 64_000, 256_000, 1_024_000)
346    scale_methods = {k: methods[k] for k in ("linear", "yesterday", "cubic_spline", "pchip",
347                                             "hybrid", "harmonic")}
348    times, exps = P.scaling(scale_methods, sizes, repeats=reps,
349                            max_n={"harmonic": 64_000 if cfg.quick else 256_000})
350    _save_table(times, cfg, "perf_scaling")
351    _save_table(exps, cfg, "perf_scaling_exponents")
352    plots.scaling_plot(times, exps, cfg.figures / "09_scaling.png")
353
354    gap_tabs, slopes = [], {}
355    for name in ("harmonic", "hybrid"):
356        t, slope = P.gap_count_scaling(methods[name], repeats=reps)
357        gap_tabs.append(t.assign(method=name))
358        slopes[name] = slope
359    gap_tab = _save_table(pd.concat(gap_tabs, ignore_index=True), cfg, "perf_gap_count_scaling")
360    plots.gap_count_plot(gap_tab, slopes, cfg.figures / "09b_gap_count_scaling.png")
361
362    prof = "\n".join([
363        "=== harmonic (all gaps), primary sensor ===", P.profile_call(methods["harmonic"], frame),
364        "=== hybrid, primary sensor ===", P.profile_call(methods["hybrid"], frame)])
365    (cfg.root / "results" / "profile_harmonic.txt").write_text(prof)
366    misc = {"vectorisation": vec, "gap_count_slopes": slopes}
367    (cfg.root / "results" / "perf_misc.json").write_text(json.dumps(misc, indent=2))
368    return {"costs": costs, "solver": solver, "exponents": exps, "optimisation": opt,
369            "gap_count": gap_tab, **misc}

373# Stage 5: reconstruct real gaps
374# ----------------------------------------------------------------------------
375def reconstruct(cfg: Config, ctx: Context, params: dict, summaries: dict | None = None) -> dict:
376    """Fill the real gaps of both sensors with the deployed method.
377
378    Output columns: ``time``, ``temperature`` (measured or reconstructed),
379    ``reconstructed`` (bool), ``gap_len_slots`` (0 if measured) and
380    ``expected_abs_error`` - the validated MAE of the selected method for
381    that gap length on that sensor, interpolated on log gap length (NaN
382    beyond the longest tested length, where the fill is unvalidated).
383    """
384    hk = _harmonic_kw(params)
385    name = params["deployed"]
386    method = method_set(hk, params["switch_len"])[name]
387    if summaries is None:
388        summaries = {}
389        for sid, fname in ((cfg.primary, "test_results_primary"),
390                           (cfg.secondary, "test_results_secondary")):
391            path = cfg.tables / f"{fname}.csv"
392            if path.is_file():
393                summaries[sid] = pd.read_csv(path)
394
395    out = {}
396    for sid, frame in ctx.frames.items():
397        filled = method(frame)
398        missing = frame["temperature"].isna().to_numpy()
399        gap_len = np.zeros(len(frame), dtype=int)
400        starts, lengths = gap_runs(missing)
401        for s, L in zip(starts, lengths):
402            gap_len[s:s + L] = L
403        expected = np.full(len(frame), np.nan)
404        expected[~missing] = 0.0
405        if sid in summaries:
406            g = summaries[sid]
407            g = g[g["method"] == name].sort_values("gap_len")
408            inside = missing & (gap_len <= g["gap_len"].max())
409            expected[inside] = np.interp(np.log(gap_len[inside]), np.log(g["gap_len"]), g["MAE"])
410        result = pd.DataFrame({"time": frame["time"], "temperature": filled,
411                               "reconstructed": missing, "gap_len_slots": gap_len,
412                               "expected_abs_error": expected})
413        result.to_csv(cfg.root / "results" / f"reconstructed_{sid}.csv", index=False)
414        out[sid] = {"filled_slots": int(missing.sum()), "gaps": int(len(lengths)),
415                    "longest_gap_h": round(float(lengths.max()) / 6, 1),
416                    "slots_beyond_validated_range": int(np.isnan(expected).sum())}
417        if lengths.max() >= 144:
418            s = starts[np.argmax(lengths)]
419            plots.reconstruction(frame, filled, s, int(lengths.max()),
420                                 cfg.figures / f"10_reconstruction_{sid}.png",
421                                 plots.LABELS.get(name, name))
422    return out

428def _md(df: pd.DataFrame, floatfmt: str = "{:.3f}") -> str:
429    """Render a DataFrame as a Markdown table (no extra dependency)."""
430    cols = [str(c) for c in df.columns]
431    lines = ["| " + " | ".join(cols) + " |", "|" + "---|" * len(cols)]
432    for row in df.itertuples(index=False):

def tune( cfg: Config, ctx: Context) -> dict: 
185def tune(cfg: Config, ctx: Context) -> dict:
186    """Tune the harmonic model and apply the selection rules on development data.
187
188    Sweeps use the partially observed design at 3, 6 and 12 h, where the
189    harmonic settings matter, and ``min_len=6`` so real single dropouts are
190    not refitted each time (this does not affect the scored gaps).
191    """
192    frame = ctx.frames[cfg.primary]
193    dev, _ = _regions(len(frame), cfg)
194    sweep_lengths = (18, 36, 72)
195
196    def sweep(param, values, base):
197        tabs = []
198        for v in values:
199            kw = {**base, param: v}
200            pg = run_gap_experiment(frame, {v: (lambda f, kw=kw: M.fill_harmonic(f, min_len=6, **kw))},
201                                    sweep_lengths, cfg.n_gaps, cfg.tune_repeats, dev, cfg.seed)
202            tabs.append(summarise(pg).assign(**{param: v}))
203        tab = pd.concat(tabs, ignore_index=True)
204        best = tab.groupby(param)["MAE"].mean().idxmin()
205        return tab, best
206
207    k_tab, k = sweep("k_daily", (0, 1, 2, 3, 4, 6), {"k_weekly": 2, "window_days": 7.0})
208    j_tab, j = sweep("k_weekly", (0, 1, 2, 3), {"k_daily": int(k), "window_days": 7.0})
209    w_tab, w = sweep("window_days", (2.0, 3.0, 5.0, 7.0, 10.0, 14.0, 21.0),
210                     {"k_daily": int(k), "k_weekly": int(j)})
211    hk = {"k_daily": int(k), "k_weekly": int(j), "window_days": float(w)}
212    for name, tab in [("tune_k_daily", k_tab), ("tune_k_weekly", j_tab), ("tune_window", w_tab)]:
213        _save_table(tab.drop(columns="method"), cfg, name)
214    plots.sensitivity(k_tab, "k_daily", cfg.figures / "05_sens_k_daily.png",
215                      "Daily harmonics (development period)", "Number of daily harmonics K")
216    plots.sensitivity(w_tab, "window_days", cfg.figures / "06_sens_window.png",
217                      "Fitting window (development period)", "Window half-width (days)")
218
219    # Method comparison on development data, then the switch rule.
220    singles = evaluate(frame, method_set(hk), dev, cfg, cfg.seed + 10)
221    switch = choose_switch(singles["combined"], "linear", "harmonic", cfg.switch_tolerance)
222    # Same seeds -> same gap placements, so the hybrid is a paired comparison.
223    hyb = evaluate(frame, {"hybrid": lambda f: M.fill_hybrid(f, switch_len=switch, **hk)},
224                   dev, cfg, cfg.seed + 10)
225    dev_comb = pd.concat([singles["combined"], hyb["combined"]], ignore_index=True)
226    _save_table(dev_comb, cfg, "dev_method_comparison")
227
228    adopted, best_single = adopt_hybrid(dev_comb, "hybrid", cfg.resolution)
229    if adopted:
230        final = "hybrid"
231    else:
232        final = simplest_within(singles["combined"], M.COMPLEXITY_ORDER, cfg.switch_tolerance)
233
234    # Optimisation rule (stage 7): if the final method is the harmonic model,
235    # deploy it with short-gap skipping when that keeps accuracy within the
236    # switch tolerance at every length.
237    piv = dev_comb.pivot_table(index="gap_len", columns="method", values="MAE")
238    hybrid_ok = bool((piv["hybrid"] <= (1 + cfg.switch_tolerance) * piv["harmonic"]).all())
239    deployed = "hybrid" if final in ("harmonic", "hybrid") and hybrid_ok else final
240
241    score = dev_comb.groupby("method")["MAE"].mean().rename("mean_dev_MAE").reset_index()
242    params = {**hk, "switch_len": switch, "best_single": best_single,
243              "hybrid_adopted_on_accuracy": adopted, "final_method": final,
244              "hybrid_accuracy_matches_harmonic": hybrid_ok, "deployed": deployed,
245              "dev_scores": score.round(4).to_dict("records")}
246    _params_path(cfg).write_text(json.dumps(params, indent=2))
247    return params

364        "=== harmonic (all gaps), primary sensor ===", P.profile_call(methods["harmonic"], frame),

def reconstruct( cfg: Config, ctx: Context, params: dict, summaries: dict | None = None) -> dict: 
376def reconstruct(cfg: Config, ctx: Context, params: dict, summaries: dict | None = None) -> dict:
377    """Fill the real gaps of both sensors with the deployed method.
378
379    Output columns: ``time``, ``temperature`` (measured or reconstructed),
380    ``reconstructed`` (bool), ``gap_len_slots`` (0 if measured) and
381    ``expected_abs_error`` - the validated MAE of the selected method for
382    that gap length on that sensor, interpolated on log gap length (NaN
383    beyond the longest tested length, where the fill is unvalidated).
384    """
385    hk = _harmonic_kw(params)
386    name = params["deployed"]
387    method = method_set(hk, params["switch_len"])[name]
388    if summaries is None:
389        summaries = {}
390        for sid, fname in ((cfg.primary, "test_results_primary"),
391                           (cfg.secondary, "test_results_secondary")):
392            path = cfg.tables / f"{fname}.csv"
393            if path.is_file():
394                summaries[sid] = pd.read_csv(path)
395
396    out = {}
397    for sid, frame in ctx.frames.items():
398        filled = method(frame)
399        missing = frame["temperature"].isna().to_numpy()
400        gap_len = np.zeros(len(frame), dtype=int)
401        starts, lengths = gap_runs(missing)
402        for s, L in zip(starts, lengths):
403            gap_len[s:s + L] = L
404        expected = np.full(len(frame), np.nan)
405        expected[~missing] = 0.0
406        if sid in summaries:
407            g = summaries[sid]
408            g = g[g["method"] == name].sort_values("gap_len")
409            inside = missing & (gap_len <= g["gap_len"].max())
410            expected[inside] = np.interp(np.log(gap_len[inside]), np.log(g["gap_len"]), g["MAE"])
411        result = pd.DataFrame({"time": frame["time"], "temperature": filled,
412                               "reconstructed": missing, "gap_len_slots": gap_len,
413                               "expected_abs_error": expected})
414        result.to_csv(cfg.root / "results" / f"reconstructed_{sid}.csv", index=False)
415        out[sid] = {"filled_slots": int(missing.sum()), "gaps": int(len(lengths)),
416                    "longest_gap_h": round(float(lengths.max()) / 6, 1),
417                    "slots_beyond_validated_range": int(np.isnan(expected).sum())}
418        if lengths.max() >= 144:
419            s = starts[np.argmax(lengths)]
420            plots.reconstruction(frame, filled, s, int(lengths.max()),
421                                 cfg.figures / f"10_reconstruction_{sid}.png",
422                                 plots.LABELS.get(name, name))
423    return out
